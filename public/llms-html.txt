# befui for HTML and htmx

> Every befui component is a custom element in a plain HTML page: one CSS file and one script, no React in your app. Build them with `npm run build:html` (dist/html/befui.css and befui.js).

```html
<link rel="stylesheet" href="/befui.css">
<script src="/befui.js" defer></script>
<bef-button variant="soft">Save</bef-button>
```

## Rules
- Tag = `bef-` + kebab-case export name: Button -> <bef-button>, DataTable -> <bef-data-table>, DialogTrigger -> <bef-dialog-trigger>. Icons too: <bef-home-icon>.
- Props are attributes in kebab-case (default-value, as-child). A bare attribute is true. Strings are the default.
- JSON props use a colon: :value="40", :items='[{"id":"a","title":"T"}]'. Inside that JSON, a string that starts with a tag becomes content: {"icon":"<bef-home-icon></bef-home-icon>"}. A string like "~2026-12-31" becomes a Date.
- A child with slot="name" becomes the prop `name` (<bef-button slot="action">Undo</bef-button>). Other children are the component's children.
- Dates as plain attributes: ~to="2026-12-31T09:00".
- Callbacks: emit="value-change open-change" fires bubbling events bef:value-change and bef:open-change (value in event.detail). The tag also writes the new value back to itself (:value, :open...), so controlled components work with no script; set :open="true" to open a controlled dialog.
- Props that are functions or JSX (render, columns, onSubmit, onApply) cannot be attributes: set them from script, element.props = { columns: [...], onSubmit: fn }.
- style works on the outer tag. Do not invent Tailwind classes in HTML (they are not in the CSS); use inline style for layout.
- Works with htmx: elements added by a swap upgrade by themselves and hx-* attributes inside are processed. Radix inputs submit with the surrounding <form>.
- Dark mode: the dark class on <html>. Theme: the same CSS variables (--primary, ...).
- window.befui.toast.success('Saved') shows a toast when <bef-toaster> is on the page.

# Examples

## Accordion (accordion)
Collapsible sections with a smooth height animation. Single or multiple open.
```html
<bef-accordion class="w-full max-w-lg" :default-open='["a"]' :items='[
  {"id":"a","title":"How are leave balances calculated?","content":"Accrual runs on the first of every month, pro-rated for new joiners."},
  {"id":"b","title":"Can managers delegate approvals?","content":"Yes. Set a delegate for any date range from the approvals page."},
  {"id":"c","title":"Where is payroll data stored?","content":"Encrypted at rest, in the region you pick at signup."}]'></bef-accordion>
```

## Activity feed (activity-feed)
Who did what and when, grouped under Today and Yesterday, with live relative times and an optional quoted detail.
```html
<bef-activity-feed class="w-full max-w-lg" :items='[
  {"id":"1","actor":"Aarav Mehta","action":"moved","target":"Payroll run","at":"2026-10-13T09:30:00"},
  {"id":"2","actor":"Diya Rao","action":"commented on","target":"Offer letter","at":"2026-10-12T16:10:00","detail":"Please double check the joining date."},
  {"id":"3","actor":"Rohan Das","action":"approved","target":"Leave request","at":"2026-10-12T11:00:00"}]'></bef-activity-feed>
```

## Alert dialog (alert-dialog)
Confirm before something destructive. Cancel and confirm, controlled by you.
```html
<bef-button id="ad-open" variant="outline">Delete project</bef-button>
<bef-alert-dialog id="ad" emit="open-change" title="Delete this project?" description="This cannot be undone." confirm-label="Delete" destructive></bef-alert-dialog>
<script>
  const ad = document.getElementById('ad');
  document.getElementById('ad-open').addEventListener('click', () => ad.setAttribute('open', 'true'));
  ad.addEventListener('bef:open-change', (e) => e.detail ? ad.setAttribute('open', 'true') : ad.removeAttribute('open'));
  ad.props = { onConfirm: () => ad.removeAttribute('open') };
</script>
```

```html
</div>
```

## Chat (chat)
Message bubbles with avatars, timestamps, a typing indicator and a composer. Enter sends.
```html
<bef-chat id="chat" me="Aarav" style="width:360px;height:380px" placeholder="Write a message"></bef-chat>
<script>
  const chat = document.getElementById('chat');
  const msgs = [{ id: '1', from: 'Diya', text: 'Did the offer go out?', at: new Date() }];
  const draw = () => chat.props = { messages: msgs, onSend: (text) => { msgs.push({ id: String(msgs.length + 1), from: 'Aarav', text, at: new Date() }); draw(); } };
  draw();
</script>
```

## Tooltip (tooltip)
Hint on hover and focus. Optional title, keyboard shortcut, light style and arrow, plus an (i) InfoTip and a TruncatedText that only shows a tooltip when cut off.
```html
<bef-tooltip-provider>
  <bef-tooltip content="Save your work" title="Save" :shortcut='["⌘","S"]'><bef-button variant="outline">Hover me</bef-button></bef-tooltip>
  <bef-tooltip content="Light card style" variant="light" arrow side="bottom"><bef-button variant="outline">Light</bef-button></bef-tooltip>
  <bef-info-tip content="Extra help text"></bef-info-tip>
</bef-tooltip-provider>
```

## Tour (tour)
Spotlight a part of the page and step through it with a card, for product walkthroughs.
```html
<div style="display:flex;gap:12px;align-items:center"><bef-button id="tour-a">Step one target</bef-button><bef-button id="tour-b" variant="outline">Step two target</bef-button><bef-button id="tour-go" variant="secondary">Start tour</bef-button></div>
<bef-tour id="tour" emit="open-change" :steps='[{"target":"#tour-a","title":"First","body":"This is the first thing."},{"target":"#tour-b","title":"Second","body":"And this is the second.","side":"bottom"}]'></bef-tour>
<script>document.getElementById('tour-go').addEventListener('click', () => document.getElementById('tour').setAttribute(':open', 'true'));</script>
```

## Tree table (tree-table)
Table whose rows have child rows. Expand and collapse branches by click or arrow keys.
```html
<bef-tree-table id="tt" class="w-full" :default-expanded='["eng"]'></bef-tree-table>
<script>
  const lakh = (n) => '₹' + (n / 100000).toFixed(1) + 'L';
  document.getElementById('tt').props = {
    rows: [
      { id: 'eng', name: 'Engineering', owner: 'Aarav Mehta', budget: 9000000, spent: 6400000, children: [
        { id: 'eng-web', name: 'Web platform', owner: 'Ananya K.', budget: 4000000, spent: 3100000, children: [
          { id: 'eng-web-fe', name: 'Frontend', owner: 'Ananya K.', budget: 2200000, spent: 1800000 }, { id: 'eng-web-be', name: 'Backend', owner: 'Kabir Shah', budget: 1800000, spent: 1300000 }] },
        { id: 'eng-mobile', name: 'Mobile', owner: 'Rohan Das', budget: 3000000, spent: 2100000 }, { id: 'eng-infra', name: 'Infrastructure', owner: 'Kabir Shah', budget: 2000000, spent: 1200000 }] },
      { id: 'sales', name: 'Sales', owner: 'Vikram Joshi', budget: 5000000, spent: 5300000, children: [
        { id: 'sales-in', name: 'India', owner: 'Vikram Joshi', budget: 3000000, spent: 3400000 }, { id: 'sales-me', name: 'Middle East', owner: 'Isha Nair', budget: 2000000, spent: 1900000 }] },
      { id: 'ops', name: 'People Ops', owner: 'Isha Nair', budget: 1500000, spent: 900000 }],
    columns: [
      { key: 'name', header: 'Department' }, { key: 'owner', header: 'Owner' },
      { key: 'budget', header: 'Budget', align: 'end', render: (r) => lakh(r.budget) }, { key: 'spent', header: 'Spent', align: 'end', render: (r) => lakh(r.spent) },
      { key: 'status', header: 'Status', render: (r) => r.spent > r.budget ? '<bef-badge variant="danger">Over</bef-badge>' : '<bef-badge variant="success">On track</bef-badge>' }],
  };
</script>
```

## Uptime bar (uptime-bar)
Status-page strip with one tick per day in green, amber or red, the uptime percentage and an incident note on hover.
```html
<bef-uptime-bar id="ub" name="API" class="w-full max-w-xl"></bef-uptime-bar>
<script>document.getElementById('ub').props = { days: Array.from({ length: 60 }, (_, i) => ({ date: '2026-08-' + String((i % 28) + 1).padStart(2, '0'), status: i === 12 ? 'down' : i === 40 ? 'degraded' : 'up' })) };</script>
```

## Virtual list (virtual-list)
Windowed list that renders only the visible rows. 100,000 items stay smooth.
```html
<div style="width:100%;max-width:32rem">
  <p style="margin:0 0 8px;font-size:12px;color:var(--muted-foreground)">100,000 rows, about 20 in the DOM at a time.</p>
  <bef-virtual-list id="vl" :row-height="44" aria-label="Employees"></bef-virtual-list>
</div>
<script>
  const depts = ['Design', 'Sales', 'Finance', 'Ops'];
  document.getElementById('vl').props = {
    items: Array.from({ length: 100000 }, (_, i) => ({ id: i + 1, name: 'Employee ' + (i + 1), dept: depts[i % 4] })), getKey: (r) => r.id,
    renderRow: (r) => `<div class="flex h-full items-center justify-between border-b px-4 text-sm"><span><span class="me-3 tabular-nums text-muted-foreground">#${r.id}</span>${r.name}</span><span class="text-muted-foreground">${r.dept}</span></div>`,
  };
</script>
```
